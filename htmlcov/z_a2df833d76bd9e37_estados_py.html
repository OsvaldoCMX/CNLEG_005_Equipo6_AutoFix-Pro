1"""Estados de la Orden de Servicio, del Trabajo y de la Solicitud de Compra. 

2 

3Los nombres son exactamente los del Glosario (sección «Estados del sistema»). 

4Las transiciones viven en una tabla separada para que OrdenDeServicio no crezca 

5con condicionales por cada estado (hallazgo SOLID H6). 

6""" 

7from __future__ import annotations 

8 

9from enum import Enum 

10 

11from autofix.dominio.compartido.errores import TransicionDeEstadoNoPermitida 

12 

13 

14class EstadoOrden(Enum): 

15 RECIBIDA = "Recibida" 

16 EN_DIAGNOSTICO = "En Diagnóstico" 

17 COTIZADA = "Cotizada" 

18 AUTORIZADA = "Autorizada" 

19 EN_REPARACION = "En Reparación" 

20 EN_ESPERA_DE_REFACCION = "En Espera de Refacción" 

21 PAUSADA_POR_RETIRO_DE_VEHICULO = "Pausada por Retiro de Vehículo" 

22 LISTA_PARA_ENTREGA = "Lista para Entrega" 

23 PAGADA = "Pagada" 

24 POR_COBRAR_CREDITO = "Por Cobrar (Crédito)" 

25 CERRADA = "Cerrada" 

26 CERRADA_SIN_REPARACION = "Cerrada sin Reparación" 

27 

28 

29class EstadoTrabajo(Enum): 

30 PENDIENTE = "Pendiente" 

31 EN_PROGRESO = "En Progreso" 

32 COMPLETADO = "Completado" 

33 NO_AUTORIZADO = "No Autorizado" 

34 

35 

36class EstadoSolicitud(Enum): 

37 SOLICITADA = "Solicitada" 

38 DISPONIBLE = "Disponible" 

39 

40 

41EO = EstadoOrden 

42 

43TRANSICIONES_DE_ORDEN: dict[EstadoOrden, set[EstadoOrden]] = { 

44 EO.RECIBIDA: {EO.EN_DIAGNOSTICO}, 

45 EO.EN_DIAGNOSTICO: {EO.COTIZADA}, 

46 EO.COTIZADA: {EO.AUTORIZADA, EO.CERRADA_SIN_REPARACION}, 

47 EO.AUTORIZADA: {EO.EN_REPARACION, EO.EN_ESPERA_DE_REFACCION}, 

48 EO.EN_REPARACION: {EO.EN_ESPERA_DE_REFACCION, EO.LISTA_PARA_ENTREGA}, 

49 EO.EN_ESPERA_DE_REFACCION: {EO.AUTORIZADA, EO.EN_REPARACION, EO.PAUSADA_POR_RETIRO_DE_VEHICULO}, 

50 EO.PAUSADA_POR_RETIRO_DE_VEHICULO: {EO.EN_ESPERA_DE_REFACCION, EO.AUTORIZADA}, 

51 EO.LISTA_PARA_ENTREGA: {EO.PAGADA, EO.POR_COBRAR_CREDITO}, 

52 EO.PAGADA: {EO.CERRADA}, 

53 EO.POR_COBRAR_CREDITO: {EO.CERRADA}, 

54 EO.CERRADA: set(), 

55 EO.CERRADA_SIN_REPARACION: set(), 

56} 

57 

58 

59def validar_transicion(actual: EstadoOrden, nuevo: EstadoOrden) -> None: 

60 if nuevo not in TRANSICIONES_DE_ORDEN[actual]: 

61 raise TransicionDeEstadoNoPermitida( 

62 f"La orden no puede pasar de «{actual.value}» a «{nuevo.value}»." 

63 )
